def cli_json_scalar(s: readbuf, o: mutbuf, w: mutbuf) -> u32:
    if length(w) != 1024 or get(w, 0) != 3552288231079560008 or get(w, 112) != 0:
        return 4294967293
    if get(w, 56) != 0:
        put(w, 56, 0)
        put(w, 24, 0)
        put(w, 8, 900)
    elif get(w, 24) != 0:
        if get(w, 32) != get(w, 16) or get(w, 40) != get(w, 64):
            return 4294967293
        put(w, 24, 0)
    elif get(w, 8) != 0:
        return 4294967293
    if get(w, 8) == 0:
        if get(w, 120) == 0 or get(w, 128) > 3 or get(w, 136) != choose(get(w, 128) == 0, 1, 0):
            return 4294967293
        if get(w, 320) != 0 or get(w, 48) != 0:
            return 4294967293
        literals(w)
        put(w, 232, 0)
        put(w, 8, 1)
    while True:
        phase = get(w, 8)
        if phase < 100:
            status = observe_scalar(w, phase)
        elif phase < 200:
            status = integer_policy(w, phase)
        elif phase < 300:
            status = dynamic_decimal(w, phase)
        elif phase < 500:
            status = materialize(s, o, w, phase)
        else:
            status = unwind(w, phase)
        if status != 4:
            return status
    return 4294967293

def observe_scalar(w: mutbuf, phase: u64) -> u64:
    if phase == 1:
        return request(w, 46, get(w, 232), 0, 0, 0, 0, 2)
    if phase == 2:
        if answer(w, 0) != 0:
            return 4294967293
        cell = get(w, 232) + 1
        put(w, 232, cell)
        if cell < 8:
            put(w, 8, 1)
            return 4
        return request(w, 17, 0, 0, 0, 0, 0, 3)
    if phase == 3:
        put(w, 200, answer(w, 4))
        put(w, 208, answer(w, 5))
        kind = get(w, 128)
        if kind == 0:
            return request(w, 38, get(w, 120), 0, 0, 0, 0, 10)
        if kind == 1:
            return request(w, 69, get(w, 120), 0, 0, 0, 0, 20)
        if kind == 2:
            return request(w, 39, get(w, 120), 0, 0, 0, 0, 30)
        return request(w, 55, get(w, 120), 0, 0, 0, 0, 40)
    if phase == 10:
        if get(w, 48) != 0:
            return failed(w)
        count = answer(w, 0)
        width = answer(w, 1)
        if width != 1 and width != 2 and (width != 4):
            return 4294967293
        if count > 9223372036854775807 - 64 >> 2:
            return request(w, 20, 0, 0, 0, 0, 0, 900)
        put(w, 144, count)
        put(w, 152, width)
        put(w, 176, 64 + count * 4)
        return request(w, 210, get(w, 176), 0, 0, 0, 0, 400)
    if phase == 20:
        if get(w, 48) != 0:
            return failed(w)
        count = answer(w, 0)
        sign = answer(w, 1)
        shift = answer(w, 2)
        width = answer(w, 4)
        if (shift != 15 or width != 2) and (shift != 30 or width != 4):
            return 4294967293
        if sign > 2 or (count == 0 and sign != 1) or (count != 0 and sign == 1):
            return 4294967293
        put(w, 144, count)
        put(w, 168, sign)
        put(w, 160, shift)
        put(w, 152, width)
        if count >= choose(shift == 15, 144, 73):
            return request(w, 70, 0, 0, 0, 0, 0, 100)
        put(w, 8, 110)
        return 4
    if phase == 30:
        if get(w, 48) != 0:
            return failed(w)
        put(w, 144, answer(w, 0))
        put(w, 176, 72)
        return request(w, 210, 72, 0, 0, 0, 0, 400)
    if phase == 40:
        if get(w, 48) != 0:
            return failed(w)
        bits = answer(w, 0)
        if bits & 448 == 0:
            return 4294967293
        put(w, 144, choose(bits & 64 != 0, 0, choose(bits & 256 != 0, 1, 2)))
        put(w, 176, 64)
        return request(w, 210, 64, 0, 0, 0, 0, 400)
    return 4294967293

def integer_policy(w: mutbuf, phase: u64) -> u64:
    if phase == 100:
        limit = answer(w, 0)
        if limit > 2147483647 or answer(w, 1) != 640 or answer(w, 2) != get(w, 160) or (answer(w, 3) != 1):
            return 4294967293
        put(w, 192, limit)
        if limit > 0:
            count = get(w, 144)
            if count >= 11:
                divisor = choose(get(w, 160) == 15, 45, 90)
                if divide_small(limit, divisor) <= divide_small(count - 11, 10):
                    return request(w, 73, limit, 0, 0, 0, 0, 900)
        put(w, 8, 110)
        return 4
    if phase == 110:
        if get(w, 144) > 1000:
            put(w, 8, 200)
            return 4
        words = get(w, 144) * get(w, 160) + 31 >> 5
        put(w, 176, 64 + words * 4)
        return request(w, 210, get(w, 176), 0, 0, 0, 0, 400)
    if phase == 120:
        limit = answer(w, 0)
        if limit > 2147483647 or answer(w, 1) != 640 or answer(w, 2) != get(w, 160) or (answer(w, 3) != 1):
            return 4294967293
        if limit > 0 and get(w, 184) - choose(get(w, 168) == 2, 1, 0) > limit:
            return request(w, 73, limit, 0, 0, 0, 0, 900)
        put(w, 8, 430)
        return 4
    return 4294967293

def dynamic_decimal(w: mutbuf, phase: u64) -> u64:
    if phase == 200:
        return request(w, 71, 512, 0, 0, 0, 0, 201)
    if phase == 201:
        if get(w, 320) == 0:
            return failed(w)
        return request(w, 44, 0, 0, 0, 0, 0, 202)
    if phase == 202:
        put(w, 216, answer(w, 0))
        return request(w, 72, get(w, 216), 544, get(w, 120), 0, 0, 203)
    if phase == 203:
        if get(w, 320) == 0:
            return failed(w)
        return request(w, 44, 1, 0, 0, 0, 0, 204)
    if phase == 204:
        put(w, 224, answer(w, 0))
        return request(w, 55, get(w, 224), 0, 0, 0, 0, 205)
    if phase == 205:
        if answer(w, 0) & 1 == 0:
            return request(w, 10, get(w, 208), 608, 51, 0, 0, 900)
        return request(w, 74, 1, 0, 0, 0, 0, 206)
    if phase == 206:
        return request(w, 45, 0, 0, 0, 0, 0, 207)
    if phase == 207:
        return request(w, 45, 1, 0, 0, 0, 0, 440)
    return 4294967293

def materialize(s: readbuf, o: mutbuf, w: mutbuf, phase: u64) -> u64:
    if phase == 400:
        if get(w, 48) != 0:
            return failed(w)
        if length(o) < get(w, 176):
            return 4294967293
        status = packet(s, o, w)
        if status != 0:
            return status
        return request(w, 211, get(w, 176), 0, 0, 0, 0, 420)
    if phase == 420:
        if get(w, 48) != 0:
            return failed(w)
        if answer(w, 0) != answer(w, 1):
            return 4294967293
        if answer(w, 0) == 4:
            return request(w, choose(get(w, 128) == 0, 76, 20), 0, 0, 0, 0, 0, 900)
        if answer(w, 0) != 0 or answer(w, 2) > 9223372036854775807 >> 2:
            return 4294967293
        put(w, 184, answer(w, 2))
        if get(w, 128) == 1 and answer(w, 2) > 640:
            return request(w, 70, 0, 0, 0, 0, 0, 120)
        put(w, 8, 430)
        return 4
    if phase == 430:
        put(o, 16, 1)
        return request(w, 212, get(w, 176), get(w, 184), 0, 0, 0, 431)
    if phase == 431:
        if get(w, 48) != 0:
            return failed(w)
        if answer(w, 0) != 0 or answer(w, 1) != 0 or answer(w, 2) != get(w, 184):
            return 4294967293
        operation = choose(get(w, 128) == 1 or get(w, 128) == 2, 75, 68)
        return request(w, operation, 0, get(w, 184), 0, 0, 0, 440)
    if phase == 440:
        if get(w, 48) != 0:
            return failed(w)
        if get(w, 320) == 0:
            return 4294967293
        put(w, 112, 1)
        return 0
    return 4294967293

def packet(s: readbuf, o: mutbuf, w: mutbuf) -> u64:
    index = 0
    while index < 64:
        store(o, index, 0)
        index = index + 1
    put(o, 0, 3555684583843054408)
    kind = get(w, 128)
    put(o, 8, kind)
    if kind == 0:
        count = get(w, 144)
        width = get(w, 152)
        if width != 1 and width != 2 and (width != 4):
            return 4294967293
        width_shift = choose(width == 1, 0, choose(width == 2, 1, 2))
        if count > length(s) >> width_shift or length(s) != count * width:
            return 4294967293
        put(o, 32, count)
        index = 0
        while index < count:
            value = storage_word(s, index * width, width)
            if value > 1114111:
                return 4294967293
            put32(o, 64 + index * 4, value)
            index = index + 1
        return 0
    if kind == 1:
        return long_packet(s, o, w)
    if kind == 2:
        put(o, 32, 1)
        put(o, 64, get(w, 144))
        return 0
    put(o, 40, get(w, 144))
    return 0

def long_packet(s: readbuf, o: mutbuf, w: mutbuf) -> u64:
    count = get(w, 144)
    width = get(w, 152)
    shift = get(w, 160)
    if count > 1000 or length(s) != count * width:
        return 4294967293
    index = 0
    words = 0
    bits = 0
    carry = 0
    while index < count:
        digit = storage_word(s, index * width, width)
        if digit >= 1 << shift or (index + 1 == count and digit == 0):
            return 4294967293
        carry = carry | digit << bits
        bits = bits + shift
        if bits >= 32:
            put32(o, 64 + words * 4, carry & 4294967295)
            words = words + 1
            carry = carry >> 32
            bits = bits - 32
        index = index + 1
    if carry != 0:
        put32(o, 64 + words * 4, carry)
        words = words + 1
    put(o, 32, words)
    put(o, 40, choose(get(w, 168) == 2, 1, 0))
    put(w, 176, 64 + words * 4)
    return 0

def unwind(w: mutbuf, phase: u64) -> u64:
    if phase == 900:
        if get(w, 320) != 0:
            return request(w, 6, 0, 0, 0, 0, 0, 901)
        put(w, 8, 901)
        return 4
    if phase == 901:
        return request(w, 46, 0, 0, 0, 0, 0, 902)
    if phase == 902:
        if answer(w, 0) != 0:
            return request(w, 45, 0, 0, 0, 0, 0, 903)
        put(w, 8, 903)
        return 4
    if phase == 903:
        return request(w, 46, 1, 0, 0, 0, 0, 904)
    if phase == 904:
        if answer(w, 0) != 0:
            return request(w, 45, 1, 0, 0, 0, 0, 905)
        put(w, 8, 905)
        return 4
    if phase == 905:
        if get(w, 320) != 0 or get(w, 48) == 0:
            return 4294967293
        put(w, 112, 2)
        return 1
    return 4294967293

def failed(w: mutbuf) -> u64:
    put(w, 8, 900)
    return 4

def request(w: mutbuf, operation: u64, a: u64, b: u64, c: u64, d: u64, e: u64, following: u64) -> u64:
    sequence = get(w, 16)
    if sequence == 18446744073709551615:
        return 4294967293
    put(w, 16, sequence + 1)
    put(w, 24, 1)
    put(w, 64, operation)
    put(w, 72, a)
    put(w, 72 + 8, b)
    put(w, 72 + 16, c)
    put(w, 72 + 24, d)
    put(w, 72 + 32, e)
    put(w, 8, following)
    return 2

def answer(w: mutbuf, index: u64) -> u64:
    return get(w, 256 + index * 8)

def divide_small(value: u64, divisor: u64) -> u64:
    quotient = 0
    remainder = 0
    remaining = 64
    while remaining != 0:
        remaining = remaining - 1
        remainder = remainder << 1 | value >> remaining & 1
        quotient = quotient << 1
        if remainder >= divisor:
            remainder = remainder - divisor
            quotient = quotient | 1
    return quotient

def storage_word(s: readbuf, at: u64, width: u64) -> u64:
    value = 0
    index = 0
    while index < width:
        value = value | byte(s, at + index) << index * 8
        index = index + 1
    return value

def get(w: mutbuf, at: u64) -> u64:
    value = 0
    index = 0
    while index < 8:
        value = value | byte(w, at + index) << index * 8
        index = index + 1
    return value

def put(w: mutbuf, at: u64, value: u64) -> u64:
    index = 0
    while index < 8:
        store(w, at + index, value >> index * 8 & 255)
        index = index + 1
    return 0

def put32(w: mutbuf, at: u64, value: u64) -> u64:
    index = 0
    while index < 4:
        store(w, at + index, value >> index * 8 & 255)
        index = index + 1
    return 0
