def cli_json_cgraph(s: readbuf, o: mutbuf, w: mutbuf) -> u32:
    if length(w) < 4096 + 256 or get(w, 0) != 3553981410266858312 or get(w, 120) != 0:
        return 4294967293
    aborting = get(w, 56)
    if aborting != 0:
        put(w, 56, 0)
        put(w, 24, 0)
        if get(w, 8) == 0:
            put(w, 208, 0)
            put(w, 120, 2)
            return 1
        fput(w, 8, 900)
    elif get(w, 24) != 0:
        if get(w, 32) != get(w, 16) or get(w, 40) != get(w, 64):
            return 4294967293
        put(w, 24, 0)
    elif get(w, 8) != 0:
        return 4294967293
    if get(w, 8) == 0:
        if get(w, 128) == 0 or get(w, 152) == 0 or get(w, 160) == 0:
            return 4294967293
        if get(w, 136) != 31404 and get(w, 136) != 31410:
            return 4294967293
        literals(w)
        put(w, 200, 4096)
        clear_frame(w, 4096)
        fput(w, 0, get(w, 128))
        fput(w, 8, 1)
        put(w, 8, 1)
    if get(w, 48) != 0 and (get(w, 64) == 202 or get(w, 64) == 204):
        fput(w, 8, 900)
    while True:
        phase = fget(w, 8)
        if phase < 20:
            status = scalar_node(s, o, w, phase)
        elif phase < 40:
            status = sequence_node(o, w, phase)
        elif phase < 80:
            status = mapping_node(o, w, phase)
        elif phase < 100:
            status = default_node(w, phase)
        elif phase < 200:
            status = marker_node(w, phase)
        else:
            status = unwind_node(w, phase)
        if status != 4:
            return status
        if get(w, 200) == 0:
            return get(w, 120) - 1
    return 4294967293

def unwind_node(w: mutbuf, phase: u64) -> u64:
    if phase == 800:
        return pop(w, 0)
    if phase != 900:
        return 4294967293
    if get(w, 192) != 0:
        return request(w, 6, 0, 0, 0, 0, 0, 900)
    flags = fget(w, 104)
    if fget(w, 16) == 3 and flags & 64 != 0:
        return request(w, 67, field(w, 104), 64, 0, 0, 0, 900)
    if flags & 16 != 0:
        return drop(w, 88, 16, 900)
    if flags & 8 != 0:
        return drop(w, 80, 8, 900)
    if flags & 4 != 0:
        return drop(w, 56, 4, 900)
    if flags & 32 != 0:
        return drop(w, 96, 32, 900)
    if fget(w, 16) == 1:
        if flags & 1 != 0:
            return drop(w, 40, 1, 900)
        if flags & 2 != 0:
            return drop(w, 48, 2, 900)
    else:
        if flags & 2 != 0:
            return drop(w, 48, 2, 900)
        if flags & 1 != 0:
            return drop(w, 40, 1, 900)
    if flags & 64 != 0:
        return request(w, 67, field(w, 104), 64, 0, 0, 0, 900)
    if flags != 0:
        return 4294967293
    return pop(w, 1)

def marker(w: mutbuf, following: u64) -> u64:
    fput(w, 128, following)
    if get(w, 144) == 0:
        fput(w, 8, following)
        return 4
    return request(w, 57, fget(w, 0), 0, 0, 0, 0, 110)

def unmark(w: mutbuf, following: u64) -> u64:
    fput(w, 128, following)
    if fget(w, 104) & 1 == 0:
        fput(w, 8, following)
        return 4
    return request(w, 60, get(w, 144), fget(w, 40), 0, 0, 0, 120)

def failed(w: mutbuf) -> u64:
    fput(w, 8, 900)
    return 4

def leave_success(w: mutbuf) -> u64:
    if fget(w, 104) & 64 != 0:
        return request(w, 67, field(w, 104), 64, 0, 0, 0, 800)
    return pop(w, 0)

def adopt(w: mutbuf, slot: u64, mask: u64, phase: u64) -> u64:
    return request(w, 62, field(w, slot), field(w, 104), mask, 0, 0, phase)

def drop(w: mutbuf, slot: u64, mask: u64, phase: u64) -> u64:
    return request(w, 64, field(w, slot), field(w, 104), mask, 0, 0, phase)

def reserve(o: mutbuf, w: mutbuf, count: u64, phase: u64) -> u64:
    current = get(w, 208)
    if current > 2305843009213693951 or count > 2305843009213693951 - current:
        return request(w, 20, 0, 0, 0, 0, 0, 900)
    required = (current + count) * 4
    if required > length(o):
        return request(w, 202, required, 0, 0, 0, 0, phase)
    return 4

def push(w: mutbuf, object: u64, level: u64, following: u64, retry: u64) -> u64:
    top = get(w, 200)
    if top > 9223372036854775807 - 256 * 2:
        return request(w, 20, 0, 0, 0, 0, 0, 900)
    next = top + 256
    if next + 256 > length(w):
        return request(w, 204, next + 256, 0, 0, 0, 0, retry)
    fput(w, 8, following)
    put(w, 200, next)
    clear_frame(w, next)
    fput(w, 0, object)
    fput(w, 24, level)
    fput(w, 8, 1)
    return 4

def pop(w: mutbuf, status: u64) -> u64:
    if fget(w, 104) != 0:
        return 4294967293
    top = get(w, 200)
    if top == 4096:
        put(w, 200, 0)
        put(w, 120, status + 1)
        if status != 0:
            put(w, 208, 0)
        return 4
    if top < 4096 + 256:
        return 4294967293
    put(w, 200, top - 256)
    fput(w, 120, status)
    return 4

def clear_frame(w: mutbuf, at: u64) -> u64:
    fill_range(w, at, 256, 0)
    return 0

def request(w: mutbuf, operation: u64, a: u64, b: u64, c: u64, d: u64, e: u64, phase: u64) -> u64:
    if get(w, 24) != 0 or get(w, 16) == 18446744073709551615:
        return 4294967293
    put(w, 16, get(w, 16) + 1)
    put(w, 64, operation)
    put(w, 72, a)
    put(w, 72 + 8, b)
    put(w, 72 + 16, c)
    put(w, 72 + 24, d)
    put(w, 72 + 32, e)
    fput(w, 8, phase)
    put(w, 24, 1)
    return 2

def field(w: mutbuf, offset: u64) -> u64:
    return get(w, 200) + offset

def fget(w: mutbuf, offset: u64) -> u64:
    return get(w, field(w, offset))

def fput(w: mutbuf, offset: u64, value: u64) -> u64:
    return put(w, field(w, offset), value)

def answer(w: mutbuf, index: u64) -> u64:
    return get(w, 256 + index * 8)

def point(o: mutbuf, index: u64, value: u64) -> u64:
    store(o, index * 4, value & 255)
    store(o, index * 4 + 1, value >> 8 & 255)
    store(o, index * 4 + 2, value >> 16 & 255)
    store(o, index * 4 + 3, value >> 24 & 255)
    return 0

def append_point(o: mutbuf, w: mutbuf, value: u64) -> u64:
    cursor = get(w, 208)
    point(o, cursor, value)
    put(w, 208, cursor + 1)
    return 0

def newline(o: mutbuf, w: mutbuf, level: u64) -> u64:
    append_point(o, w, 10)
    index = 0
    while index < level * 2:
        append_point(o, w, 32)
        index = index + 1
    return 0

def append_constant(o: mutbuf, w: mutbuf, which: u64) -> u64:
    if which == 0:
        append_point(o, w, 110)
        append_point(o, w, 117)
        append_point(o, w, 108)
        append_point(o, w, 108)
    elif which == 1:
        append_point(o, w, 116)
        append_point(o, w, 114)
        append_point(o, w, 117)
        append_point(o, w, 101)
    else:
        append_point(o, w, 102)
        append_point(o, w, 97)
        append_point(o, w, 108)
        append_point(o, w, 115)
        append_point(o, w, 101)
    return 0

def get(w: mutbuf, at: u64) -> u64:
    value = 0
    index = 0
    while index < 8:
        value = value | byte(w, at + index) << index * 8
        index = index + 1
    return value

def put(w: mutbuf, at: u64, value: u64) -> u64:
    index = 0
    while index < 8:
        store(w, at + index, value >> index * 8 & 255)
        index = index + 1
    return 0
