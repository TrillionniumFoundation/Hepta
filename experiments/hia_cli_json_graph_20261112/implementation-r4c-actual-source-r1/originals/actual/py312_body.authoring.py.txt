def field(s: readbuf, pc: u64, at: u64) -> u64:

def read(s: readbuf, at: u64) -> u64:

    index = 0
    while index < 8:

def get(w: mutbuf, at: u64) -> u64:

def put(w: mutbuf, at: u64, value: u64) -> u64:
