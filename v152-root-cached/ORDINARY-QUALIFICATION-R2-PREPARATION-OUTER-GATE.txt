def main():
    rr,_=read(REVIEW);review=json.loads(rr)
    assert pin(rr)==REVIEW_PIN and review['status']=='PASS_ROOT_STATIC_ORDINARY_QUALIFICATION_R2_PATH_PROJECTION_ONLY'
    assert review['source_files']==3225 and review['owner_count']==18 and review['method_invocations']==85 and review['required_builds']==9
    assert review['binders']==review['products']==review['source_copies']==0
    lr,ls=read(LAUNCH);launch=json.loads(lr)
    assert pin(lr)==LAUNCH_PIN and review['launch']==pin(lr)
    pp=Path(launch['plan']['path']);pr,_=read(pp);plan=json.loads(pr)
    assert pin(pr)==PLAN_PIN and review['plan']==pin(pr)
    assert plan['state']=='AUTHORING_ONLY_AWAITING_ROOT_AUTHORIZATION' and plan['source_files']==3225
    assert plan['source_tree']=='6c506654638513dbe10623da9458d3904a208cc8'
    assert plan['products_executed']==0 and not plan['source_copy_performed']
    def check(d):
        b,s=read(Path(d['path']))
        assert pin(b)=={k:d['pin'][k] for k in ('bytes','git_blob','sha256')}
        assert s==[d['stat'][k] for k in ('st_dev','st_ino','st_mode','st_nlink','st_size','st_mtime_ns','st_ctime_ns')]
        return b
    for d in plan['controls'].values():check(d)
    for d in plan['expected_runtime'].values():check(d)
    assert check(launch['plan'])==pr and check(launch['parent'])==check(plan['controls']['prepare_qualification.py'])
    assert launch['command']==[plan['runtime']['launch_executable'],'-I','-B',plan['controls']['prepare_qualification.py']['path'],'--plan',str(pp),'--plan-sha256',pin(pr)['sha256']]
    assert launch['exact_clean_env_argv']==['/usr/bin/env','-i']+[k+'='+launch['environment'][k] for k in sorted(launch['environment'])]+launch['command']
    assert launch['qualification_seconds']==180 and launch['cleanup_seconds']==3 and launch['automatic_retry'] is False
    assert launch['cwd']==str(BASE) and launch['outer_directory']==str(OUT)
    assert not os.path.lexists(plan['scope']) and not os.path.lexists(OUT)
    for d in plan['scope_ancestors']:
        s=Path(d['path']).lstat()
        assert stat.S_ISDIR(s.st_mode) and {'path':d['path'],'dev':s.st_dev,'ino':s.st_ino,'mode':s.st_mode,'uid':s.st_uid,'gid':s.st_gid}==d
    fs=os.statvfs(Path(plan['scope']).parent)
    assert fs.f_bavail*fs.f_frsize>=plan['storage_floor']==1073741824 and not fs.f_flag&os.ST_NOEXEC
    launch['argv']=launch['exact_clean_env_argv']
    OUT.mkdir(mode=0o700)
    dump('INTENT.json',{'status':'ROOT_AUTHORIZED_ORDINARY_R2_DATA_PREPARATION_ONLY_ONCE','review':pin(rr),'launch':pin(lr),'launch_seven_stats':ls,'argv':launch['argv'],'cwd':launch['cwd'],'outer_watchdog_seconds':180,'cleanup_seconds':3,'maximum_attempts':1,'source_copy_files':0,'binding_source_files':3225,'generator_calls':0,'compile_calls':0,'native_calls':0,'vm_calls':0,'tests':0,'qualification_authorized':False,'whole_native':False})
