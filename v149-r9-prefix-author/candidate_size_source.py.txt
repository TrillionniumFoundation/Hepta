

def candidate_source_bytes():
    """Explicit R9 input binding; all original R8 input/history guards remain intact."""
    binding_raw, binding_mode, _ = read_regular(ROOT / 'CANDIDATE-BINDING.json')
    require(pin(binding_raw, binding_mode) == PLAN['controls']['author:CANDIDATE-BINDING.json']['identity'],
            'new candidate binding differs')
    binding = json.loads(binding_raw)
    require(binding['schema'] == 'hia.R9.actual-generated-candidate-binding.v1'
            and binding['generation_calls'] == binding['check_calls'] == 1
            and binding['old_SOURCE_rebound'] is False
            and binding['old_source_tree'] == PLAN['source']['tree'],
            'new candidate generation provenance differs')
    material = {}
    for name, row in binding['files'].items():
        raw, mode, stamp = read_regular(row['path'])
        require(pin(raw, mode) == row['identity']
                and [stamp[key] for key in ('st_dev','st_ino','st_mode','st_nlink','st_size','st_mtime_ns','st_ctime_ns')]
                    == row['seven_stats'],
                'generated candidate input changed: ' + name)
        material[name] = raw
    require(material['generate_report'] == material['check_report'], 'actual generation/check report mismatch')
    report = json.loads(material['generate_report'])
    proof = report['generated']['word64_projection']
    require(set(report['generated']) == {'functions', 'blocks', 'ops', 'word64_projection'}
            and {key: report['generated'][key] for key in ('functions','blocks','ops')}
                == {'functions':107,'blocks':1456,'ops':6347}
            and proof['original_counts'] == {'functions':107,'blocks':1456,'ops':6664}
            and proof['after_counts'] == {'functions':107,'blocks':1456,'ops':6347}
            and len(proof['removed_identity_casts']) == 317
            and proof['fixed_word_bits'] == 64
            and proof['whole_inverse_equal'] is True
            and proof['complete_original_Core_verifier_returned'] is True
            and proof['all_other_operations_kept_in_order'] is True
            and proof['external_f0_unchanged'] is True
            and proof['public_compiler_or_VM_changed'] is False,
            'new exact Core projection proof differs')
    for stage in ('generate', 'check'):
        result = json.loads(material[stage+'_result'])
        owner = json.loads(material[stage+'_owner_returned'])
        complete_owner(owner, 180)
        require(result['status'] == 'PASS_ORIGINAL_R9_GENERATOR_STAGE'
                and result['stage'] == stage and result['finalization_errors'] == []
                and result['frozen_source_runtime_equal'] is True,
                'new generation stage is incomplete')
    raw = material['source']
    require(len(raw) == report['source_bytes'] and pin(raw)['sha256'] == report['source_sha256']
            and 0 < len(raw) <= CAPACITY, 'actual new source pin differs')
    require(pin(input_bytes('R8.hepta'))['sha256']
            == '83953e66677fa0cd7f63f3b1742276d954537df71487c3cd4a6eef9486e3e4de',
            'original R8 preimage history differs')
    return raw
