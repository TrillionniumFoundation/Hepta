"""Behavioral contracts for the learned discrete reset/validity candidate."""
import copy
import unittest
import torch
from .p2_reset_validity import (RawResetValidityMemory,hard_event,reset_without_rebinding_batch,
                                RESET_VALIDITY_MODES,MAPPED_MODE)
from .p2_associative_memory import delta_batch,DELTA_VOCAB_SIZE,interpret_delta
from .p2_round2_raw_tokens import UNKNOWN


class ResetValidityContracts(unittest.TestCase):
    def setUp(self):
        torch.set_num_threads(1);torch.manual_seed(61113)
        self.model=RawResetValidityMemory('hcsn_reset_validity').double()
        self.batch=delta_batch(820003091,n=4,kinds=('mqar','overwrite','state_tracking','multi_reset'))

    def test_exact_erasure_of_all_history_for_detected_reset(self):
        # Force a model-produced event via learned parameters, never token IDs.
        with torch.no_grad():
            self.model.control_logits.weight.zero_();self.model.control_logits.bias.zero_()
            self.model.channel_bias[1]=30.
        tokens=self.batch.tokens[:,:1];dt=self.batch.duration[:,:1].double()
        a=(torch.randn(4,8,16,dtype=torch.double),torch.randn(4,16,dtype=torch.double),
           torch.randn(4,16,dtype=torch.double),torch.randn(4,16,dtype=torch.double))
        b=tuple(x*17+31 for x in a)
        with torch.no_grad():
            x,sx,trace=self.model(tokens,dt,a,return_state=True,return_trace=True)
            y,sy=self.model(tokens,dt,b,return_state=True)
        self.assertTrue(bool((trace['events'][...,1]==1).all()))
        for left,right in zip(sx,sy):torch.testing.assert_close(left,right,rtol=0,atol=0)
        for i in (0,1,3):self.assertEqual(float(sx[i].abs().max()),0.)
        torch.testing.assert_close(x,y,rtol=0,atol=0)
        self.assertTrue(bool((x.argmax(-1)==UNKNOWN).all()))
        self.assertTrue(bool((x.exp()[...,UNKNOWN]>.997).all()))

    def test_inactive_write_cannot_accumulate_false_validity(self):
        with torch.no_grad():
            self.model.control_logits.weight.zero_();self.model.control_logits.bias.zero_()
            self.model.channel_bias[:]=torch.tensor([-30.,-30.],dtype=torch.double)
        tokens=self.batch.tokens[:,:64];dt=self.batch.duration[:,:64].double()
        with torch.no_grad():x,state,trace=self.model(tokens,dt,return_state=True,return_trace=True)
        self.assertEqual(float(state[0].abs().max()),0.)
        self.assertEqual(float(state[1].abs().max()),0.)
        self.assertTrue(bool((trace['events']==0).all()))
        self.assertTrue(bool((x.argmax(-1)==UNKNOWN).all()))

    def test_straight_through_values_are_exact_and_gradients_finite(self):
        p=torch.tensor([0.,.01,.49,.5,.9,1.],dtype=torch.double,requires_grad=True)
        event=hard_event(p)
        self.assertEqual(event.detach().tolist(),[0.,0.,0.,1.,1.,1.])
        event.sum().backward();torch.testing.assert_close(p.grad,torch.ones_like(p),rtol=0,atol=0)

    def test_causal_streaming_and_batch_independence(self):
        b=self.batch;k=19
        with torch.no_grad():
            whole=self.model(b.tokens,b.duration.double())
            first,state=self.model(b.tokens[:,:k],b.duration[:,:k].double(),return_state=True)
            second=self.model(b.tokens[:,k:],b.duration[:,k:].double(),state)
            solo=self.model(b.tokens[:1],b.duration[:1].double())
        torch.testing.assert_close(torch.cat((first,second),1),whole,rtol=1e-12,atol=1e-12)
        torch.testing.assert_close(solo,whole[:1],rtol=1e-12,atol=1e-12)

    def test_semantic_token_permutation_equivariance(self):
        perm=torch.randperm(DELTA_VOCAB_SIZE);clone=copy.deepcopy(self.model)
        with torch.no_grad():
            clone.embedding.weight[perm]=self.model.embedding.weight
            x=self.model(self.batch.tokens,self.batch.duration.double())
            y=clone(perm[self.batch.tokens],self.batch.duration.double())
        torch.testing.assert_close(x,y,rtol=0,atol=0)

    def test_no_targets_enter_forward(self):
        with torch.no_grad():x=self.model(self.batch.tokens,self.batch.duration.double())
        self.batch.targets.fill_(0);self.batch.stale_answers.fill_(0)
        with torch.no_grad():y=self.model(self.batch.tokens,self.batch.duration.double())
        torch.testing.assert_close(x,y,rtol=0,atol=0)

    def test_probability_normalization_and_real_query_gradients(self):
        scores=self.model(self.batch.tokens,self.batch.duration.double())
        torch.testing.assert_close(scores.exp().sum(-1),torch.ones_like(scores[...,0]),rtol=1e-12,atol=1e-12)
        torch.nn.functional.cross_entropy(scores.flatten(0,1),self.batch.targets.flatten()).backward()
        for p in self.model.parameters():
            self.assertIsNotNone(p.grad);self.assertTrue(bool(torch.isfinite(p.grad).all()))
        self.assertGreater(float(self.model.address.weight.grad.norm()),0.)
        self.assertGreater(float(self.model.control_logits.weight.grad.norm()),0.)

    def test_mapped_put_linear_values_and_gradients(self):
        mapped=RawResetValidityMemory(MAPPED_MODE).double();mapped.load_state_dict(self.model.state_dict())
        x=self.model(self.batch.tokens,self.batch.duration.double())
        y=mapped(self.batch.tokens,self.batch.duration.double())
        torch.testing.assert_close(x,y,rtol=1e-10,atol=1e-10)
        x.square().mean().backward();y.square().mean().backward()
        for a,b in zip(self.model.parameters(),mapped.parameters()):
            torch.testing.assert_close(a.grad,b.grad,rtol=1e-8,atol=1e-8)

    def test_independent_interpreter_and_reset_only_stress(self):
        batch=reset_without_rebinding_batch(820003092,n=3,cycles=2,queries_per_reset=8)
        for row,L in enumerate(batch.lengths):
            self.assertEqual(interpret_delta(batch.tokens[row,:L].tolist()),batch.targets[row,:L].tolist())
        valid=batch.targets!=-100
        self.assertEqual(int(valid.sum()),3*(4+2*8))
        self.assertEqual(int((batch.targets==UNKNOWN).sum()),3*2*8)

    def test_paired_parameter_initialization_and_invalid_inputs(self):
        values=[]
        for mode in RESET_VALIDITY_MODES:
            torch.manual_seed(613);model=RawResetValidityMemory(mode)
            values.append(torch.cat([p.detach().flatten() for p in model.parameters()]))
            self.assertEqual(values[-1].numel(),2276)
        torch.testing.assert_close(values[0],values[1],rtol=0,atol=0)
        with self.assertRaises(ValueError):self.model(self.batch.tokens,self.batch.duration.double()*0)
        with self.assertRaises(ValueError):self.model(self.batch.tokens+100,self.batch.duration.double())
        with self.assertRaises(ValueError):self.model(self.batch.tokens,self.batch.duration.double(),initial=())


if __name__=='__main__':unittest.main()
