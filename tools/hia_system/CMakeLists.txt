cmake_minimum_required(VERSION 3.20)
project(hepta_hia_system LANGUAGES C CXX)
find_package(LLVM 20 REQUIRED CONFIG)
if(NOT LLVM_PACKAGE_VERSION VERSION_EQUAL "20.1.8")
  message(FATAL_ERROR "hepta.hia.system.v1 requires LLVM 20.1.8")
endif()
add_executable(hia-system system_module.cpp system_instructions.cpp)
target_compile_features(hia-system PRIVATE cxx_std_17)
target_include_directories(hia-system SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
llvm_map_components_to_libnames(HIA_LLVM_LIBS Core IRReader BitWriter Support)
target_link_libraries(hia-system PRIVATE ${HIA_LLVM_LIBS})
